    <script>
        const sb = supabase.createClient(CONFIG.supabaseUrl, CONFIG.supabaseAnon);

        async function init() {
            const loadingText = document.getElementById('loading-text');
            
            try {
                // 1. Check Auth
                const { data: { session } } = await sb.auth.getSession();
                if (!session) {
                    window.location.href = '/?login=true&redirect=/guide/natchez-lower';
                    return;
                }

                // 2. Check Purchase
                loadingText.textContent = 'Verifying Purchase...';
                const response = await fetch('/.netlify/functions/check-purchase', {
                    method: 'POST',
                    headers: { 
                        'Content-Type': 'application/json',
                        'Authorization': `Bearer ${session.access_token}`
                    },
                    body: JSON.stringify({ route_slug: 'natchez-lower' })
                });
                
                const { purchased } = await response.json();
                if (!purchased) {
                    window.location.href = '/natchez-trace-lower.html';
                    return;
                }

                // 3. Load Data & Render
                loadingText.textContent = 'Loading Guide Content...';
                const guideRes = await fetch('./guide-data.json');
                const guide = await guideRes.json();
                
                render(guide);

                // 4. Show Content
                document.getElementById('loading').style.opacity = '0';
                setTimeout(() => {
                    document.getElementById('loading').style.display = 'none';
                    document.getElementById('guide-content').style.display = 'block';
                }, 500);

            } catch (err) {
                console.error('Error:', err);
                loadingText.innerHTML = `<span style="color:red">Error loading guide. Please try refreshing.</span>`;
            }
        }

        // ── RENDERING ENGINE (Adapted from guide-template.html) ──
        function el(tag, cls, html) {
            const e = document.createElement(tag);
            if (cls) e.className = cls;
            if (html) e.innerHTML = html;
            return e;
        }

        function formatTownName(key) {
            if (!key) return '—';
            return key.replace(/_/g, ' ').split(' ').map(w => w.charAt(0).toUpperCase() + w.slice(1)).join(' ');
        }

        function pageShell(routeLabel, pageNum, pageLabel) {
            const page = el('div', 'page');
            const hdr = el('div', 'page-header');
            hdr.innerHTML = `<span class="page-header-route">${routeLabel}</span><span class="page-header-logo">Open<span>Ride</span>.bike</span>`;
            page.appendChild(hdr);
            const body = el('div', 'page-body');
            page.appendChild(body);
            const ftr = el('div', 'page-footer');
            ftr.innerHTML = `<span>${routeLabel}</span><span>${pageNum}</span>`;
            page.appendChild(ftr);
            return { page, body };
        }

        function sectionHead(label) {
            return `<div class="section-head"><span class="section-label">${label}</span><div class="section-head-line"></div></div>`;
        }

        function render(guide) {
            const container = document.getElementById('guide-content');
            const r = guide.route;
            let pageNum = 1;

            // Simple render for web
            guide.days.forEach(day => {
                const isArrival = day.miles === '0 miles riding' || day.miles === 0;
                const { page, body } = pageShell(r.name, `Page ${pageNum++}`, `Day ${day.number}`);
                
                const hero = el('div', 'day-hero');
                hero.innerHTML = `
                    <div class="day-hero-img-overlay"></div>
                    <div class="day-num">0${day.number}</div>
                    <div class="day-hero-content">
                        <div class="day-title">${day.title}</div>
                        <div class="day-meta">
                            <div class="day-meta-item"><span class="day-meta-val">${isArrival ? 'Arrival' : day.miles}</span><span class="day-meta-lbl">Distance</span></div>
                            <div class="day-meta-item"><span class="day-meta-val">${formatTownName(day.overnight_town)}</span><span class="day-meta-lbl">Overnight</span></div>
                        </div>
                    </div>`;
                body.appendChild(hero);
                body.appendChild(el('p', 'day-description', day.description));

                if (day.poi && day.poi.length) {
                    const poiSec = el('div');
                    poiSec.innerHTML = sectionHead('Points of Interest');
                    const list = el('div', 'day-description');
                    day.poi.forEach(p => {
                        list.innerHTML += `<p><strong>MP ${p.milepost}: ${p.name}</strong><br>${p.description}</p>`;
                    });
                    poiSec.appendChild(list);
                    body.appendChild(poiSec);
                }

                if (day.lodging && day.lodging.length) {
                    const lodgSec = el('div');
                    lodgSec.innerHTML = sectionHead('Lodging');
                    const grid = el('div', 'lodging-grid');
                    day.lodging.forEach(l => {
                        grid.innerHTML += `<div class="lodging-card ${l.highlight ? 'highlight' : ''}">
                            <div class="lodging-name">${l.name}</div>
                            <div style="font-size:0.9rem; color:var(--accent); margin:0.3rem 0;">${l.price || ''}</div>
                            <div style="font-size:0.85rem; color:#666;">${l.notes || ''}</div>
                        </div>`;
                    });
                    lodgSec.appendChild(grid);
                    body.appendChild(lodgSec);
                }

                container.appendChild(page);
            });
        }

        init();
    </script>
